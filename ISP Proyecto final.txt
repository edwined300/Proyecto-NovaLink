SANTO DOMINGO

enable
configure terminal

interface Ethernet0/3
 description Link_to_ISP
 ip address 8.0.0.2 255.255.255.240
 duplex full
 ip nat outside
 no shutdown
 exit

interface Ethernet0/0
 ip nat inside
 exit
interface Ethernet0/1
 ip nat inside
 exit

ip route 0.0.0.0 0.0.0.0 8.0.0.1
access-list 100 permit ip 172.16.0.0 0.0.255.255 any
ip nat inside source list 100 interface Ethernet0/3 overload

router ospf 1
 default-information originate
 exit
end
write
